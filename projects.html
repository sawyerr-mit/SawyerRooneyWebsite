def companion_matrix(coeffs):
    coeffs = np.asarray(coeffs, dtype=float)

    if len(coeffs) < 2:
        raise ValueError("Polynomial must have degree >= 1.")

    coeffs = coeffs / coeffs[0]
    n = len(coeffs) - 1

    C = np.zeros((n, n))
    C[1:, :-1] = np.eye(n - 1)
    C[:, -1] = -coeffs[1:][::-1]
